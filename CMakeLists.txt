cmake_minimum_required(VERSION 3.16)
project(cst_audio_12dyn VERSION 0.1.0 LANGUAGES CXX)
set(CMAKE_CXX_STANDARD 17)
set(CMAKE_CXX_STANDARD_REQUIRED ON)
set(CMAKE_CXX_EXTENSIONS OFF)
add_library(cst_audio STATIC src/audio.cpp)
target_include_directories(cst_audio PUBLIC ${CMAKE_CURRENT_SOURCE_DIR}/include)
if(MSVC)
  target_compile_options(cst_audio PRIVATE /W4)
else()
  target_compile_options(cst_audio PRIVATE -Wall -Wextra -Wpedantic)
endif()
add_executable(cst-foley src/main.cpp)
target_link_libraries(cst-foley PRIVATE cst_audio)
include(CTest)
if(BUILD_TESTING)
  add_executable(cst_audio_tests tests/test_audio.cpp)
  target_link_libraries(cst_audio_tests PRIVATE cst_audio)
  add_test(NAME cst_audio_tests COMMAND cst_audio_tests)
endif()
